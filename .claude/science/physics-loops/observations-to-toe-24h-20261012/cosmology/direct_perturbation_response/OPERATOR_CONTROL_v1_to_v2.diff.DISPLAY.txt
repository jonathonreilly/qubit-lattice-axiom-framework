+    rows = []

+                # Differentiate the full potential gradient; do not use the
+                # simplified on-trajectory entries in linear_rhs.
+                hessian = np.column_stack([
+                    np.imag(potential(phi.astype(complex) + 1e-27j * np.eye(2)[j],
+                                      a, eta)[1]) / 1e-27

+                      Hess_error < 2e-12, max_normalized_error=float(Hess_error))
+                H = np.sqrt((2*sigma + velocity@velocity + 2*V)/3)

+                    expected = np.empty((7, 7))
+                    for j in range(7):

+                        field = basis[3:5, j]
+                        field_dot = H*basis[5:7, j]

+                        theta_dot = -2*H*theta + K2*(P+alpha@field)
+                        field_ddot = (-3*H*field_dot - K2*field - hessian@field

+                                      + 4*velocity*P_dot

+                                      - alpha*sigma*d)
+                        expected[:, j] = np.r_[
+                            P_dot/H, -T+3*P_dot/H,

+                                     maximum_normalized_error=maximum,
+                                     expected=expected.tolist(), actual=actual.tolist()))
