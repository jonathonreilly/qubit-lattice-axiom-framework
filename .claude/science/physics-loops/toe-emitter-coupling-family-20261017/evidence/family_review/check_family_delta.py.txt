def q(z):
 z=F(z);return {'exact':str(z),'decimal_display_only':float(z)}
def sq(z):return s.Rational(z.numerator,z.denominator)
cert=json.loads((root/'initial_sources/two_body_interval_bridge/CERTIFICATE.json').read_text())
p={k:F(z) for k,z in cert['exact_parameters'].items()}

J=p['J']; zr=p['cos_k'];zi=p['sin_k']; E0=p['epsilon']
lo,hi=F(1,10**12),F(1,10**8);sig=F(1,10**6);h=sig/2;v=F(1,40);L=12
assert zr*zr+zi*zi==1 and E0==2*J*(1-zr)
c=zr*zr-zi*zi;sin2=2*zr*zi
assert c>F(3,5) and c-2*sig>F(3,5)

# The author's sharper identity is checked here after the separate stage-one
# mass derivation was frozen. The phase derivative is at fixed apparatus phi.
k=s.symbols('k',real=True);js,cs,ks=s.symbols('J c kap',positive=True,real=True)
vg=2*js*s.sin(k);Sigma=-s.I*(2-2*cs*s.exp(2*s.I*k))/vg
assert s.trigsimp(s.expand_complex(s.re(Sigma))+2*cs*s.cos(k)/js)==0
gamma=(2-2*cs*s.cos(2*k))/vg
assert s.trigsimp(s.expand_complex(-s.im(Sigma))-gamma)==0
E=2*js*(1-s.cos(k));xr=E-ks**2*(-2*cs*s.cos(k)/js)
assert s.simplify(s.diff(xr,k)-(1-ks**2*cs/js**2)*vg)==0
x,y,xp,yp,aa,V=s.symbols('x y xp yp a V',real=True)
theta_prime=(xp*y-x*yp)/(x*x+y*y)
assert s.simplify(((theta_prime+x*yp/(x*x+y*y))/aa).subs(xp,aa*V)-y*V/(x*x+y*y))==0
wrong=s.simplify(((theta_prime-x*yp/(x*x+y*y))/aa).subs(xp,aa*V)-y*V/(x*x+y*y))
assert wrong==-2*x*yp/(aa*(x*x+y*y))
amin=1-hi**2*c/J**2
assert F(1,2)<amin<1
m,Mx,G=F(1,10**8),F(1,10**7),F(52)
assert v*h-160*hi**2>m and 2*J*h+160*hi**2<Mx
assert F(32,25)/v<G
assert 1+(Mx/(19*lo**2))**2 < F(8,3)**80
Bmass=2*hi**2*G/m+2*hi**2*560*80/v
assert Bmass==F('0.0000010403584') and Bmass<3
eta_h=h/(sin2-h)
delta_m=Bmass/3+eta_h**2
assert delta_m<F(7,20000000)

# Stage one's independently executed exact Schur bounds are controlling for
# inverse and overlap; recomputed constants below feed new phase/tail tests.
r,g=F(1,7),F(9,50);delta=(1+r*r)*r**(L-1);wc=16*(1+r*r)*r**(L-2)
amax,bmin,bmax=2*sig,2*sin2-2*sig,F(2)
eta=amax/bmin;sPQ=amax*206*delta;sQP=delta*206*bmax;gamma0=bmin*bmin/(4*J)
xi=2*eta+hi**2*sPQ*6*sQP/gamma0
eta_p=(eta+6*hi**2*sPQ*wc/bmin+xi)/(1-xi)
off=6*hi*wc+6*hi**2*delta*206*bmax*(1+eta_p)
assert eta_p==F(first['upper_endpoint_Schur_overlap_checks']['eta_p_upper']['exact'])
assert off==F(first['upper_endpoint_Schur_overlap_checks']['epsilon_off_upper']['exact'])
seed_sq=2*(delta_m+eta_p+off)
seed_rounded=2*(F(7,20000000)+F(4,10**6)+F(1,10**14))
assert seed_sq<seed_rounded==F('0.00000870000002')<F(3,1000)**2

# Complete coherent reflection estimate, including the closed complement.
closed=6*hi**2*206*delta*wc+6*hi**2*delta**2*206**2*bmax*(1+eta_p)*(4*J/bmin)
phase_error=2*eta+2*(1+eta)*eta_p+closed+4*eta**2
assert 4*eta**2<F(1,4) and phase_error<F(11,10**6)
target_error=F(176,10**15)+702*sig+F(11,10**6)
assert target_error<F(714,10**6)

# Bounds are analytic positive majorants at hi, not physical matrix samples.
d=F(61,1000);a1=2/sig+13
sigmap=560+24*206*2+amax*2000*2+amax*206*24
assert sigmap<11000 and 2*J+hi**2*sigmap<d
M16p=2*J/(1-r*r)
bottomp=2*J+hi**2*(560+24*40*2+amax*100*2+amax*40*24)
Maugp=max(M16p,bottomp)+48*hi
assert Maugp<F(63,1000)
wcp=(700*(1+r*r)+32*r/24)*r**(L-2)+16*(L-2)*(1+r*r)*r**(L-3)/24
w0=bmax+amax+wc
w1=(L*bmax+2)+700*amax+(L*amax+2)+wcp
assert w0<3 and w1<30
assert w0/18<1 and d*w0/18**2+hi**2*w1/18<1
K0=332+2*492+L*(492+492)+492*F(1011,1000)
K1=7404+492+2*21468+L*(16000+(L+1)*492+21468)+14580*F(1011,1000)+492*F(1,24)*F(105,100)
assert K0==F('13621.412')<20000 and K1==F('591961.905')<10**6
assert 31*hi+1+20000*hi<2
assert (2700+10**6)*hi**3+1+20000*hi<2
assert (700+21468)*hi**2+492<1000
assert F(2,19)<1 and F(26,19)*hi**2+2*d/19**2<1
assert F(720,19)*hi**2+160*d/19**2<1
assert 2*hi**2+4*(F(720,19)*hi**2+160*d/19**2)<5
weighted_inverse=(1+6*hi**2*sPQ)*(1+6*hi**2*sQP)/(1-xi)
assert weighted_inverse<2
assert 1+20000*hi<2 and 36*hi<1
assert 374*hi+18*10**6*hi**2<1 and 6*10**6*hi<1
assert 108*d+180<187 and 72*d+60<65 and 12*d<1
U,kk,ap,dd=s.symbols('U kap a1 d',positive=True)
product=kk*(ap*U+dd*U**2)*(32+12*kk*U)+kk*U*(2701+131*kk*U+2*kk*U**2)+s.Rational(5,2)*kk*U*(32+12*kk*U)
expected=kk*(32*ap+2781)*U+(kk**2*(12*ap+161)+32*kk*dd)*U**2+kk**2*(12*dd+2)*U**3
assert s.expand(product-expected)==0
Djscaled=(236*a1+5518)*hi**2+544*d*hi+6*d+1
assert Djscaled<F('1.366000380')<F(3,2)
F1scaled=a1*hi+17*d
phase_L1scaled=F(2,500)*hi+48960*hi**2+320*d
assert F1scaled<F(2)
assert F1scaled+F(5,2)*F(1,500)*hi<3
assert F1scaled+phase_L1scaled+F(5,2)*F(1,500)*hi<23

# Alternate squared-L2 polynomials, with every positive cross term retained.
c2,c3,c4,c5,c6=F(320,19),F(160,361),F(160,6859),F(320,390963),F(120,2476099)
Rf=c2*a1**2*hi**4+2*c3*a1*d*hi**2+c4*d*d
aa1=a1+2;bb=d+2880*hi**2;cc=640*d

assert author_f==F('0.0097600010000065')<F(1,100)
assert author_out==F('0.28304000100005278')<F(3,10)

# kappa^2*T=10^6 is exact for the whole interval, not an endpoint fit.
