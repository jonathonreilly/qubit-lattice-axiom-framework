def digest(raw):
    return hashlib.sha256(raw).hexdigest()

def dump(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, indent=2, sort_keys=True) + '\n', encoding='utf-8')

def put(rel, raw):
    path = C / rel
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes(raw)

def git_bytes(rel):
    return subprocess.check_output(['git', 'show', BASE + ':' + rel], cwd=W)

def ast_text(node):
    return ast.dump(node, include_attributes=False)

def ast_hash(node):
    return digest(ast_text(node).encode())

def source_node(text, node):
    return ast.get_source_segment(text, node)

assert D.name == 'emitter_family_publication_draft'
assert not C.exists(), 'An existing overlay is not silently overwritten.'
freeze = json.loads((S / 'FREEZE.json').read_text())
fm = json.loads((S / 'FILE_MANIFEST.json').read_text())

for row in fm['files']:

        assert source['commit'] == BASE

        assert parent_raw == raw, rel
        reused[dest] = raw
        transport = 'reuse_exact_tracked_parent_file'
    else:
        dest = PREFIX + '/frozen_family/' + rel + ('.txt' if rel.endswith('.md') else '')
        put(dest, raw)
        transport = 'byte_preserved_new_copy'
    mapping.append({'source_relative_path': rel, 'path': dest, 'sha256': digest(raw),

                    'source_packet': 'emitter_resource_family_candidate'})
by_source = {row['source_relative_path']: row for row in mapping}

reused[PARENT_PRIMARY] = parent_primary

resolver_node = next(n for n in parent_tree.body if isinstance(n, ast.ClassDef) and n.name == 'FrozenInputs')
artifact_node = next(n for n in parent_tree.body if isinstance(n, ast.FunctionDef) and n.name == 'verify_artifacts')

removed_names = {'HERE', 'PRE', 'D'}
adapted = []
events = []
new_index = 0
for index, node in enumerate(tree.body):
    original = source_node(source, node)
    event = {'original_statement_index': index, 'original_lines': [node.lineno, node.end_lineno],
             'original_ast_sha256': ast_hash(node)}
    remove = ((isinstance(node, ast.Assign) and any(isinstance(t, ast.Name) and t.id in removed_names for t in node.targets))
              or (isinstance(node, ast.For) and original.startswith("for row in PRE['bindings']"))
              or (isinstance(node, ast.Assert) and original.startswith('assert resource.getrlimit')))
    if remove:
        event.update(kind='removed_historical_launch_transport', original_text=original,
                     reason='Historical source/prelaunch paths and fixed old launch assertion are replaced by declared local hash checking and actual main-process cap metadata.')
    elif isinstance(node, ast.Assign) and any(isinstance(t, ast.Name) and t.id == 'data' for t in node.targets):
        replacement = "data=json.loads(inputs.read_text('family_author', 'inputs/interval.json'))"
        adapted.append(replacement)
        event.update(kind='adapted_input_read', original_text=original, replacement_text=replacement,
                     new_statement_indices=[new_index], mathematical_change=False)
        new_index += 1
    elif isinstance(node, ast.Assign) and any(isinstance(t, ast.Name) and t.id == 'report' for t in node.targets):
        old = "'wall_cap_external_seconds':30"
        new = "'wall_cap_external_seconds':inputs.shared_limits()['external_wall_cap_seconds']"
        assert original.count(old) == 1
        replacement = original.replace(old, new)
        adapted.append(replacement)
        event.update(kind='actual_limit_metadata_only', original_text=old, replacement_text=new,
                     new_statement_indices=[new_index], mathematical_change=False)
        new_index += 1
    elif isinstance(node, ast.Expr) and isinstance(node.value, ast.Call) and isinstance(node.value.func, ast.Name) and node.value.func.id == 'print':
        replacement = "report['_canonical_execution'] = inputs.execution_metadata('family_author')\nreturn report"
        adapted.append(replacement)
        event.update(kind='return_instead_of_print_transport', original_text=original, replacement_text=replacement,
                     new_statement_indices=[new_index, new_index + 1], mathematical_change=False)
        new_index += 2
    else:
        adapted.append(original)
        event.update(kind='unchanged_statement_ast', new_statement_indices=[new_index])
        new_index += 1
    events.append(event)
