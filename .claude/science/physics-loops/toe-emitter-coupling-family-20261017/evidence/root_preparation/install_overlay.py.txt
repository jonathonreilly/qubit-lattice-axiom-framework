def archive(label,root,paths):
    rows=[]
    for src in sorted(paths):
        rel=str(src.relative_to(root))
        if src.is_symlink():
            target=str(src.readlink());rows.append({'original_path':rel,'kind':'symlink_fixture_metadata_only','target':target,'target_text_sha256':hashlib.sha256(target.encode()).hexdigest()});continue
        b=src.read_bytes();raw=digest(b)
        if raw['sha256'] in known:
            p,compression,stored=known[raw['sha256']]
        else:
            # Preserve exact problematic raw whitespace losslessly, never rewrite it.
            compression='gzip' if re.search(rb'[ \t]+\r?$',b,re.M) or b.endswith(b'\n\n') else 'none'
            stored=gzip.compress(b,mtime=0) if compression=='gzip' else b
            p=Q/'evidence'/label/(rel+('.gz' if compression=='gzip' else '.txt'))
            p.parent.mkdir(parents=True,exist_ok=True);p.write_bytes(stored)
            known[raw['sha256']]=(p,compression,stored)
        rows.append({'original_path':rel,'kind':'file','path':str(p.relative_to(W)),**digest(stored),'compression':compression,'raw_bytes':raw['bytes'],'raw_sha256':raw['sha256']})
    record={'source_directory':str(root),'scope':'Lossless reconstruction map; historical paths and statuses retained, no scientific replay.', 'files':rows,'count':len(rows)}
    write(Q/'evidence'/(label.upper()+'_MAP.json'),record)
    return record
