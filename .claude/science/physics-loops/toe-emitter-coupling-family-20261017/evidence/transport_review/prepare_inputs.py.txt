#!/usr/bin/env python3
"""Finite, deduplicated input capture for a read-only static transport review."""
from pathlib import Path
import datetime,hashlib,json
R=Path(__file__).resolve().parent
N=R.parent; P=N/'emitter_family_publication_draft'; S=N/'emitter_resource_family_candidate'
W=N.parent/'correlation_worktree'
assert not (R/'INPUTS.json').exists()
plan={'declared_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),
      'kind':'independently written static AST/hash/closure checks only',
      'planned_runs':1,'maximum_runs_after_recorded_correction':2,
      'cpu_soft_hard_seconds':[15,15],'external_wall_seconds':20,
      'max_input_file_bindings':350,'max_input_bytes':20000000,
      'max_matrix_dimension':0,'scientific_runner_calls':0,'mutation_executions':0,
      'author_or_adapter_imports':0,'target_code_execution':False,
      'memory_cap':'not imposed; RSS measured',
      'failure_policy':'Preserve full source, prelaunch, streams and receipt for every attempt; no silent retries.'}
(R/'RESOURCE_PLAN.json').write_text(json.dumps(plan,indent=2)+'\n')
source_map=json.loads((P/'canonical/outputs/emitter_coupling_family_2026_10_17_inputs/SOURCE_MAP.json').read_text())
paths=set(q.absolute() for root in [P,S] for q in root.rglob('*') if q.is_file())
paths.update(W/row['path'] for row in source_map['files'] if row['transport']=='reuse_exact_tracked_parent_file')
paths.update(W/row['path'] for row in source_map['infrastructure'])
paths.update(W/rel for rel in ['docs/ai_methodology/SCIENCE_WORKFLOW.md','docs/ai_methodology/skills/SKILL_FRESHNESS_CHECK.md','docs/ai_methodology/skills/physics-claim-reviewer/SKILL.md','docs/ai_methodology/skills/review-loop/SKILL.md'])
paths.update(N/'emitter_family_root_qualification'/name for name in ['ACCEPTANCE.json','QUALIFICATION.md'])
assert len(paths)<=plan['max_input_file_bindings']
rows=[];total=0;(R/'inputs/bytes').mkdir(parents=True,exist_ok=True)
for path in sorted(paths):
    raw=path.read_bytes();total+=len(raw);assert total<=plan['max_input_bytes']
    digest=hashlib.sha256(raw).hexdigest();dest=R/'inputs/bytes'/digest
    if not dest.exists():dest.write_bytes(raw)
    assert dest.read_bytes()==raw
    rows.append({'original':str(path),'snapshot':str(dest.relative_to(R)),
                 'bytes':len(raw),'sha256':digest,'original_is_symlink':path.is_symlink(),
                 'original_resolved_path':str(path.resolve())})
index={'created_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),
       'draft_root':str(P),'family_root':str(S),'provided_parent_worktree':str(W),
       'provided_parent_commit':'0f1022e03dbbed8085035f2a378be8ae1aca02e8',
       'parent_commit_verification':'No Git invoked per task; current bytes compared with exact frozen parent identities by the static checker.',
       'deduplication':'Identical bytes stored once by SHA256; every original path remains a separate binding.',
       'files':rows,'count':len(rows),'total_logical_input_bytes':total,
       'unique_snapshot_count':len({r['snapshot'] for r in rows})}
(R/'INPUTS.json').write_text(json.dumps(index,indent=2)+'\n')
print(json.dumps({k:index[k] for k in ['count','total_logical_input_bytes','unique_snapshot_count']},indent=2))
