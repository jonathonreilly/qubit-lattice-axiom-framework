def raw(p):return (R/bindings[str(p)]['snapshot']).read_bytes()
def obj(p):return json.loads(raw(p))
def sha(b):return hashlib.sha256(b).hexdigest()
def ad(n):return ast.dump(n,include_attributes=False)
def parse(s):return ast.parse(s).body[0]
def aid(n):return sha(ad(n).encode())
def binding(p):return {'original':str(p),'bytes':len(raw(p)),'sha256':sha(raw(p))}
