class Adapt(ast.NodeTransformer):
    def __init__(self,role):self.role=role;self.events=[]
    def event(self,kind,old,new):
        self.events.append({'kind':kind,'original_line':getattr(old,'lineno',None),
            'old_source':ast.unparse(old),'new_source':ast.unparse(new),
            'old_ast':adump(old),'new_ast':adump(new)})
        return ast.copy_location(new,old)
    def visit_Call(self,node):
        # Transform only literal file-read APIs, retaining all surrounding
        # manifest loops, byte/hash assertions, parsers and scientific use.
        if isinstance(node.func,ast.Attribute) and node.func.attr in ('read_text','read_bytes'):
            assert not node.args and not node.keywords
            recv=node.func.value; logical=None
            if isinstance(recv,ast.BinOp) and isinstance(recv.op,ast.Div) and isinstance(recv.left,ast.Name) and recv.left.id=='ROOT':
                logical=copy.deepcopy(recv.right)
            elif isinstance(recv,ast.Call) and isinstance(recv.func,ast.Name) and recv.func.id=='Path' and len(recv.args)==1:
                if isinstance(recv.args[0],ast.Name) and recv.args[0].id=='__file__':
                    replacement=ast.Call(ast.Attribute(ast.Name('inputs',ast.Load()),'original_program_bytes',ast.Load()),[ast.Constant(self.role)],[])
                    assert node.func.attr=='read_bytes'
                    return self.event('historical_self_source_read',node,replacement)
                logical=copy.deepcopy(recv.args[0])
            assert logical is not None,ast.unparse(node)
            replacement=ast.Call(ast.Attribute(ast.Name('inputs',ast.Load()),node.func.attr,ast.Load()),[ast.Constant(self.role),logical],[])
            return self.event('explicit_local_input_read',node,replacement)
        return self.generic_visit(node)
    def visit_Dict(self,node):
        original=copy.deepcopy(node)
        node=self.generic_visit(node)
        for j,key in enumerate(node.keys):
            if isinstance(key,ast.Constant) and key.value=='script_sha256':
                new=ast.Constant('historical_source_program_sha256')
                node.keys[j]=self.event('historical_program_identity_label',key,new)
        positions=[j for j,k in enumerate(node.keys) if isinstance(k,ast.Constant) and k.value in cap_keys]
        if positions:
            # Values of all non-resource keys are retained verbatim as ASTs.
            old=copy.deepcopy(node)
            historical=ast.Dict([copy.deepcopy(node.keys[j]) for j in positions],
                                [copy.deepcopy(node.values[j]) for j in positions])
            keys=[k for j,k in enumerate(node.keys) if j not in positions]
            values=[v for j,v in enumerate(node.values) if j not in positions]
            keys += [ast.Constant('historical_source_run_limits'),ast.Constant('actual_shared_process_limits')]
            values += [historical,ast.Call(ast.Attribute(ast.Name('inputs',ast.Load()),'shared_limits',ast.Load()),[],[])]
            node=self.event('resource_metadata_only',old,ast.Dict(keys,values))
        return node

def removed_kind(node,role):
    text=ast.unparse(node)
    if isinstance(node,ast.Expr) and isinstance(node.value,ast.Call):
        call=node.value
        if isinstance(call.func,ast.Attribute) and isinstance(call.func.value,ast.Name) and call.func.value.id=='resource' and call.func.attr=='setrlimit':return 'per_program_CPU_cap_replaced_by_shared_primary_cap'
        if isinstance(call.func,ast.Name) and call.func.id=='print':return 'terminal_stdout_replaced_by_return'
        if isinstance(call.func,ast.Attribute) and call.func.attr=='write_text':return 'terminal_output_write_replaced_by_return'
    if isinstance(node,ast.Assign) and any(isinstance(t,ast.Name) and t.id=='ROOT' for t in node.targets):return 'historical_filesystem_root_replaced_by_explicit_resolver'
    if role=='weighted_reviewer':
        if isinstance(node,ast.Assign) and any(isinstance(t,ast.Name) and t.id in ('parser','args','outpath') for t in node.targets):return 'output_CLI_transport_removed'
        if text.startswith('parser.add_argument(') or text.startswith('outpath.relative_to('):return 'output_CLI_transport_removed'
    return None
