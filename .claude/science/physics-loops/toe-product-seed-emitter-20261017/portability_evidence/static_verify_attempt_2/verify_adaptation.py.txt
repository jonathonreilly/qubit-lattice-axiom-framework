def sha(raw): return hashlib.sha256(raw).hexdigest()
def digest(p):
    raw = p.read_bytes()

def removed_is_transport(node, role):
    if isinstance(node, ast.Expr) and isinstance(node.value, ast.Call):
        c = node.value
        if isinstance(c.func, ast.Attribute) and isinstance(c.func.value, ast.Name):
            if c.func.value.id == 'resource' and c.func.attr == 'setrlimit':
                return True
            if role == 'weighted_reviewer' and (c.func.value.id, c.func.attr) in {('parser','add_argument'), ('outpath','relative_to')}:
                return True
        if isinstance(c.func, ast.Name) and c.func.id == 'print': return True
        if isinstance(c.func, ast.Attribute) and c.func.attr == 'write_text': return True
    if isinstance(node, ast.Assign):
        names = {t.id for t in node.targets if isinstance(t, ast.Name)}
        return bool(names) and names <= ({'ROOT','parser','args','outpath'} if role == 'weighted_reviewer' else {'ROOT'})
    return False

def check_event(event, role):
    assert event['kind'] in allowed_event_kinds
    old = ast.parse(event['old_source'], mode='eval').body
    new = ast.parse(event['new_source'], mode='eval').body
    assert adump(old) == event['old_ast'] and adump(new) == event['new_ast']
    kind = event['kind']
    if kind in {'explicit_local_input_read','historical_self_source_read'}:
        assert isinstance(old, ast.Call) and isinstance(old.func, ast.Attribute)
        assert old.func.attr in {'read_text','read_bytes'} and not old.args and not old.keywords
        assert isinstance(new, ast.Call) and isinstance(new.func, ast.Attribute)
        assert adump(new.func.value) == adump(ast.Name('inputs', ast.Load()))
        assert new.args[0].value == role and not new.keywords
        recv = old.func.value
        if kind == 'historical_self_source_read':
            assert ast.unparse(recv) == 'Path(__file__)'
            assert new.func.attr == 'original_program_bytes' and len(new.args) == 1
        else:
            assert new.func.attr == old.func.attr and len(new.args) == 2
            if isinstance(recv, ast.BinOp):
                assert isinstance(recv.op, ast.Div) and ast.unparse(recv.left) == 'ROOT'
                logical = recv.right
            else:
                assert isinstance(recv, ast.Call) and ast.unparse(recv.func) == 'Path'
                assert len(recv.args) == 1 and not recv.keywords
                logical = recv.args[0]
            assert adump(new.args[1]) == adump(logical)
    elif kind == 'historical_program_identity_label':
        assert isinstance(old, ast.Constant) and old.value == 'script_sha256'
        assert isinstance(new, ast.Constant) and new.value == 'historical_source_program_sha256'
    else:
        assert isinstance(old, ast.Dict) and isinstance(new, ast.Dict)
        positions = [i for i,k in enumerate(old.keys) if isinstance(k,ast.Constant) and k.value in cap_keys]
        assert positions
        expected = ast.Dict([k for i,k in enumerate(old.keys) if i not in positions] +
                            [ast.Constant('historical_source_run_limits'), ast.Constant('actual_shared_process_limits')],
                            [v for i,v in enumerate(old.values) if i not in positions] +
                            [ast.Dict([old.keys[i] for i in positions], [old.values[i] for i in positions]),
                             ast.parse('inputs.shared_limits()', mode='eval').body])
        assert adump(expected) == adump(new)
    return old, new

class UndoOne(ast.NodeTransformer):
    def __init__(self, old, new): self.old=old; self.new=adump(new); self.count=0
    def visit(self, node):
        if adump(node) == self.new:
            self.count += 1
            return copy.deepcopy(self.old)
        return super().visit(node)

def visible_markdown(text):
    out=[];fence=None
    for line in text.splitlines():
        m=re.match(r'^\s*(`{3,}|~{3,})',line)
        if m:
            if fence is None:fence=(m.group(1)[0],len(m.group(1)))
            elif m.group(1)[0]==fence[0] and len(m.group(1))>=fence[1]:fence=None
            continue
        if fence is None:out.append(line)
    assert fence is None
    return '\n'.join(out)
