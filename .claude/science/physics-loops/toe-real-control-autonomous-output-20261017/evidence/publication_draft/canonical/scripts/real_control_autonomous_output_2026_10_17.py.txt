class FrozenInputs:

        self.reads=[]
    def read_bytes(self, role, logical):
        row=self.table['roles'][role]['aliases'][str(logical)]
        rel=Path(row['path'])
        assert not rel.is_absolute() and '..' not in rel.parts and rel.as_posix()==row['path']
        path=self.repo/rel
        assert path.is_file() and not path.is_symlink(), row['path']
        assert not any(p.is_symlink() for p in path.parents if p!=self.repo and self.repo in p.parents)
        path.resolve().relative_to(self.repo)
        stored=path.read_bytes()
        assert len(stored)==row['bytes'] and hashlib.sha256(stored).hexdigest()==row['sha256'], row['path']
        assert row['compression'] in ('none','gzip') and 0<=row['raw_bytes']<=16777216
        raw=gzip.decompress(stored) if row['compression']=='gzip' else stored
        assert len(raw)==row['raw_bytes'] and hashlib.sha256(raw).hexdigest()==row['raw_sha256'], row['path']
        self.reads.append({'role':role,'logical':str(logical),'path':row['path'],'kind':'finite declared source/integrity input','stored_sha256':row['sha256'],'raw_sha256':row['raw_sha256']})
        return raw
    def read_text(self, role, logical):
        return self.read_bytes(role,logical).decode('utf-8')
    def original_program_bytes(self, role):
        return self.read_bytes(role,'__original_program__')

        return {'cpu_soft_hard_seconds':list(resource.getrlimit(resource.RLIMIT_CPU)), 'external_wall_cap_seconds':45, 'wall_enforcement':'Required external canonical runner; this class does not implement wall timeout.'}
