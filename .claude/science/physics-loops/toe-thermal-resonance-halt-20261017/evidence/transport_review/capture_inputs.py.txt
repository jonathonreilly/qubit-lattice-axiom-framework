def sha(b):return hashlib.sha256(b).hexdigest()
def save(p,b):
 p.parent.mkdir(parents=True,exist_ok=True)
 with p.open('wb') as f:f.write(b);f.flush();os.fsync(f.fileno())
def cp(src,dest,expected=None):
 b=src.read_bytes()
 if expected:assert sha(b)==expected,(str(src),sha(b))
 save(dest,b)
 return {'source':str(src),'snapshot':str(dest.relative_to(D)),'bytes':len(b),'sha256':sha(b)}
