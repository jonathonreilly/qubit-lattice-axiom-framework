"""Independent Pauli algebra with exact phases (C13 independent check).

A Pauli is (x, z, k) meaning i^k * prod_q X_q^{x_q} * prod_q Z_q^{z_q}
(on every qubit X is written to the left of Z). x, z are Python ints (bitsets).
"""
import itertools
import numpy as np

def pc(a):
    return bin(a).count("1")

def mul(p, q):
    x1, z1, k1 = p
    x2, z2, k2 = q
    return (x1 ^ x2, z1 ^ z2, (k1 + k2 + 2 * pc(z1 & x2)) % 4)

def dag(p):
    x, z, k = p
    return (x, z, (-k + 2 * pc(x & z)) % 4)

def is_herm(p):
    x, z, k = p
    return (k - pc(x & z)) % 2 == 0

def commutes(p, q):
    return (pc(p[0] & q[1]) + pc(p[1] & q[0])) % 2 == 0

def X(q):
    return (1 << q, 0, 0)

def Z(q):
    return (0, 1 << q, 0)

def Y(q):
    return (1 << q, 1 << q, 1)  # Y = i X Z

IDENT = (0, 0, 0)

def prod(ps):
    r = IDENT
    for p in ps:
        r = mul(r, p)
    return r

def bits(a):
    out = []
    while a:
        lb = a & -a
        out.append(lb.bit_length() - 1)
        a ^= lb
    return out

def apply_clifford(p, imgX, imgZ):
    """Conjugation by the Clifford with X_q -> imgX[q], Z_q -> imgZ[q]."""
    x, z, k = p
    r = (0, 0, k)
    for q in bits(x):
        r = mul(r, imgX[q])
    for q in bits(z):
        r = mul(r, imgZ[q])
    return r


class SignedGroup:
    """Abelian group generated by commuting Hermitian Paulis, with signs.

    Rows are actual group elements (products of generators with phases),
    stored with distinct top bits of the combined vector x | z << n.
    """

    def __init__(self, n):
        self.n = n
        self.rows = {}
        self.minus_one = False
        self.rank = 0

    def _comb(self, p):
        return p[0] | (p[1] << self.n)

    def _reduce(self, p):
        acc = p
        v = self._comb(acc)
        while v:
            t = v.bit_length() - 1
            r = self.rows.get(t)
            if r is None:
                return acc, False
            acc = mul(acc, r)
            v = self._comb(acc)
        return acc, True

    def add(self, g):
        assert is_herm(g)
        acc, inspan = self._reduce(g)
        if inspan:
            if acc[2] == 2:
                self.minus_one = True
            assert acc[2] in (0, 2)
            return False
        self.rows[self._comb(acc).bit_length() - 1] = acc
        self.rank += 1
        return True

    def status(self, p):
        """'in' if p in group, 'minus' if -p in group, 'out' otherwise."""
        acc, inspan = self._reduce(p)
        if not inspan:
            return "out"
        if acc[2] == 0:
            return "in"
        if acc[2] == 2:
            return "minus"
        return "phase"


def proper_rotations():
    mats = []
    for perm in itertools.permutations(range(3)):
        for sg in itertools.product([1, -1], repeat=3):
            M = np.zeros((3, 3), dtype=int)
            for i in range(3):
                M[i, perm[i]] = sg[i]
            if round(np.linalg.det(M)) == 1:
                mats.append(M)
    assert len(mats) == 24
    return mats
