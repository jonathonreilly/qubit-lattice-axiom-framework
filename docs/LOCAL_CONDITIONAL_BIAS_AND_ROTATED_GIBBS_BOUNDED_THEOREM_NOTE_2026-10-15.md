    (h+epsilon)I + [[-h,-epsilon],[-epsilon,h]],

    w_00=e^[-beta(h+epsilon)] [cosh(beta r)+(h/r)sinh(beta r)],

    P(X_1=1 | X_2=1)

    beta(r-epsilon) > log[2/(1-h/r)]
