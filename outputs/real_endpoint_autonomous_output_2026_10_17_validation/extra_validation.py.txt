from pathlib import Path
import ast, hashlib, json, os, resource, shutil, subprocess, sys, time
ROOT=Path(__file__).resolve().parent
W=ROOT.parents[1]/'correlation_worktree'
REL='scripts/real_endpoint_autonomous_output_2026_10_17.py'
SRC=(W/REL).read_text()
node=ast.parse(SRC)
INPUTS=next(ast.literal_eval(n.value) for n in node.body if isinstance(n,ast.Assign) and any(isinstance(t,ast.Name) and t.id=='AUDIT_INPUT_PATHS' for t in n.targets))
def identity(p):
 b=p.read_bytes();return {'path':str(p),'bytes':len(b),'sha256':hashlib.sha256(b).hexdigest()}
def caps():
 resource.setrlimit(resource.RLIMIT_CPU,(10,10));resource.setrlimit(resource.RLIMIT_FSIZE,(16*1024**2,16*1024**2))
def run(name,cmd,cwd,paths,timeout=10):
 d=ROOT/name;d.mkdir(exist_ok=True)
 pre={'command':cmd,'cwd':str(cwd),'utc':time.time(),'wall_limit_sec':timeout,'cpu_limit_sec':10,'file_limit_bytes':16*1024**2,'memory_cap':'not applied','sources':[identity(p) for p in paths]}
 with (d/'PRELAUNCH.json').open('w') as f:json.dump(pre,f,indent=2);f.flush();os.fsync(f.fileno())
 t=time.monotonic()
 with (d/'stdout.txt').open('w') as out,(d/'stderr.txt').open('w') as err:
  p=subprocess.Popen(cmd,cwd=cwd,stdout=out,stderr=err,preexec_fn=caps)
  try:code=p.wait(timeout=timeout)
  except subprocess.TimeoutExpired:p.kill();p.wait();code='timeout'
 result={'exit_code':code,'elapsed_sec':time.monotonic()-t,'outputs':[identity(d/'stdout.txt'),identity(d/'stderr.txt')]}
 (d/'RESULT.json').write_text(json.dumps(result,indent=2)+'\n')
 return code,(d/'stdout.txt').read_text(),(d/'stderr.txt').read_text()

results=[]
for name,before,after,expected in [
 ('resource_radius','F(19,2)+2*J+w+2*J+kap+Vmax,18','F(19,2)+2*J+w+2*J+kap+Vmax,17','one-head spectral radius'),
 ('load',"2*p['gA'],2","2*p['gA'],1",'old1 added load'),
 ('variance','tail=kap/(v*sig)','tail=2*kap/(v*sig)','variance tail amplitude'),
 ('historical_onset','T=10**136;M=50*T','T=10**130;M=50*T','time denominator lower squared'),
]:
 if SRC.count(before)!=1:raise RuntimeError('nonunique '+name)
 scratch=ROOT/('scratch_'+name);(scratch/'scripts').mkdir(parents=True,exist_ok=True)
 shutil.copytree(W/'outputs/real_endpoint_autonomous_output_2026_10_17_inputs',scratch/'outputs/real_endpoint_autonomous_output_2026_10_17_inputs',dirs_exist_ok=True)
 runner=scratch/REL;runner.write_text(SRC.replace(before,after))
 code,out,err=run('mutation_'+name,[sys.executable,'-O',str(runner)],scratch,[runner]+[scratch/p for p in INPUTS])
 ok=code==1 and 'PASS integrity:' in out and expected in out.split('TOTAL FAIL:')[-1]
 results.append({'name':name,'expected_failure':expected,'detected':ok,'exit_code':code})
 if not ok:raise RuntimeError(name+out[-500:]+err)
for name in ['same_byte_symlink','symlink_ancestor','directory_input','tampered_bytes']:
 scratch=ROOT/('scratch_'+name);(scratch/'scripts').mkdir(parents=True,exist_ok=True)
 shutil.copytree(W/'outputs/real_endpoint_autonomous_output_2026_10_17_inputs',scratch/'outputs/real_endpoint_autonomous_output_2026_10_17_inputs',dirs_exist_ok=True)
 runner=scratch/REL;runner.write_text(SRC)
 target=scratch/INPUTS[0]
 if name=='same_byte_symlink':
  target.unlink();target.symlink_to(W/INPUTS[0])
 elif name=='symlink_ancestor':
  directory=target.parent;shutil.rmtree(directory);directory.symlink_to((W/INPUTS[0]).parent,target_is_directory=True)
 elif name=='directory_input':target.unlink();target.mkdir()
 else:target.write_bytes(target.read_bytes()+b' ')
 code,out,err=run('mutation_'+name,[sys.executable,'-O',str(runner)],scratch,[runner])
 expected='symlink input or ancestor' if 'symlink' in name else ('input is not a regular file' if name=='directory_input' else 'integrity mismatch')
 ok=code==1 and expected in out and 'PASS arithmetic:' not in out
 results.append({'name':name,'expected_failure':expected,'detected':ok,'exit_code':code})
 if not ok:raise RuntimeError(name+out[-500:]+err)
(ROOT/'EXTRA_MUTATION_SUMMARY.json').write_text(json.dumps(results,indent=2)+'\n')
print(json.dumps({'extra_mutations':len(results),'all_intended_failures':True}))
